cd "d:\github\change radar\frontend"
npm run build

# Step 2: Deploy to Cloudflare Worker + Edge API
cd "d:\github\change radar"

# OR Deploy to Cloudflare Pages:
npx wrangler pages deploy frontend/dist --project-name change-radar

# OR Push to GitHub (Auto-deploys via Cloudflare Pages CI):
git add .

git push origin main
